# Prosistem.IA — HTTPS básico para hospedagem Apache / cPanel
# Renomeie este arquivo para ".htaccess" e coloque na mesma pasta do index.html
# (normalmente public_html). Antes, ative o SSL gratuito (AutoSSL / Let's Encrypt)
# no painel da hospedagem.

# Redireciona todo acesso HTTP para HTTPS
<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteCond %{HTTPS} !=on
    RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
</IfModule>

# Cabeçalhos de segurança
<IfModule mod_headers.c>
    Header always set Strict-Transport-Security "max-age=31536000" "expr=%{HTTPS} == 'on'"
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

# Compressão
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/css application/javascript image/svg+xml
</IfModule>
